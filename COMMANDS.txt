Run all commands from the repository root (/workspaces/mini-engine).

=== SETUP ===
pip install -r script-builder/requirements.txt
pip install -r workers/requirements-pandas.txt     # pandas worker
pip install -r workers/requirements-polars.txt     # polars worker

=== SCRIPT BUILDER ===
# Defaults: pipeline=dag/pipeline_test.json, framework=pandas, output=output/<pipeline name>/
PYTHONPATH=script-builder/src python script-builder/src/main.py

# Choose the DAG
PYTHONPATH=script-builder/src python script-builder/src/main.py dag/pipeline_clean.json

# Polars backend
PYTHONPATH=script-builder/src python script-builder/src/main.py dag/pipeline_test.json --framework polars

# Custom output dir, id, version, and also generate pipeline.ipynb
PYTHONPATH=script-builder/src python script-builder/src/main.py dag/pipeline_test.json \
  --output-dir output/my_pipeline --pipeline-id my_pipeline --pipeline-version 2 --notebook

# As a module
PYTHONPATH=script-builder/src python -m script_builder dag/pipeline_test.json

# Output: main.py and manifest.json (plus pipeline.ipynb with --notebook)

=== WORKER ===
# Defaults: artifact=output/pipeline_test, worker=framework from manifest.json, base-dir=repository root
PYTHONPATH=workers/src python workers/src/main.py

# Run a specific artifact
PYTHONPATH=workers/src python workers/src/main.py output/pipeline_test

# Force a worker
PYTHONPATH=workers/src python workers/src/main.py output/pipeline_test --worker polars

# With runtime parameters, external datasources and a base dir for relative paths
PYTHONPATH=workers/src python workers/src/main.py output/pipeline_test \
  --params-file params.json --datasources-file datasources.json --base-dir .

# params.json:      {"param_name": "value"}
# datasources.json: {"<input_port_id>": "data/Orders_Central.csv"}

# As a module
PYTHONPATH=workers/src python -m workers output/pipeline_test

=== RUN THE GENERATED SCRIPT DIRECTLY (without worker) ===
python output/pipeline_test/main.py --params-file params.json --datasources-file datasources.json --base-dir .

=== TESTS ===
python -m pytest -q script-builder/tests
python -m pytest -q workers/tests
python -m pytest -q
